Write one SQLite SELECT/WITH query for each of the following 24 tasks. Do not run queries or inspect any files, worlds, gold answers or tools. Return one JSON object only: {"version":1,"candidates":[{"id":"C01","task_id":"C01","sql":"SELECT ..."},...]}. Use each task ID once. No markdown, explanations or invented metadata.

Only these deterministic SQL functions are supported: abs, avg, coalesce, count, dense_rank, ifnull, lag, lead, length, lower, max, min, nullif, rank, round, row_number, substr, substring, sum, total, upper. SELECT/CTEs, joins, EXISTS, CASE, set operations and windows are allowed. One statement, <=16 KiB; no PRAGMA/schema access/extensions/writes. SQLite STRICT types; dates are canonical valid YYYY-MM-DD; integer currency cents. Column aliases do not matter, position and declared type do. Unordered results are bags retaining duplicates; ordered results must follow every tie breaker. Finite real results use absolute/relative tolerance 1e-9. All output columns non-NULL. IDs positive; all FKs valid. Payments and events occur on/after parent dates. Resolved dates are on/after opening. Empty tables/zero-child entities, repeated dates, NULLs and overpayments are possible.

SCHEMA commerce

CREATE TABLE customers(id INTEGER PRIMARY KEY CHECK(id>0), region TEXT) STRICT;
CREATE TABLE orders(id INTEGER PRIMARY KEY CHECK(id>0), customer_id INTEGER NOT NULL REFERENCES customers(id), status TEXT NOT NULL CHECK(status IN ('paid','pending','cancelled')), ordered_on TEXT NOT NULL) STRICT;
CREATE TABLE lines(id INTEGER PRIMARY KEY CHECK(id>0), order_id INTEGER NOT NULL REFERENCES orders(id), sku TEXT NOT NULL CHECK(sku IN ('A','B','C')), quantity INTEGER NOT NULL CHECK(quantity BETWEEN 1 AND 5), unit_cents INTEGER NOT NULL CHECK(unit_cents BETWEEN 0 AND 10000)) STRICT;

SCHEMA support

CREATE TABLE agents(id INTEGER PRIMARY KEY CHECK(id>0), team TEXT NOT NULL) STRICT;
CREATE TABLE tickets(id INTEGER PRIMARY KEY CHECK(id>0), agent_id INTEGER REFERENCES agents(id), opened_on TEXT NOT NULL, resolved_on TEXT CHECK(resolved_on IS NULL OR resolved_on>=opened_on), priority INTEGER NOT NULL CHECK(priority BETWEEN 0 AND 3)) STRICT;
CREATE TABLE events(id INTEGER PRIMARY KEY CHECK(id>0), ticket_id INTEGER NOT NULL REFERENCES tickets(id), kind TEXT NOT NULL CHECK(kind IN ('reply','note')), happened_on TEXT NOT NULL) STRICT;

SCHEMA billing

CREATE TABLE accounts(id INTEGER PRIMARY KEY CHECK(id>0), plan TEXT NOT NULL CHECK(plan IN ('basic','pro')), active INTEGER NOT NULL CHECK(active IN (0,1))) STRICT;
CREATE TABLE invoices(id INTEGER PRIMARY KEY CHECK(id>0), account_id INTEGER NOT NULL REFERENCES accounts(id), issued_on TEXT NOT NULL, total_cents INTEGER NOT NULL CHECK(total_cents BETWEEN 0 AND 50000)) STRICT;
CREATE TABLE payments(id INTEGER PRIMARY KEY CHECK(id>0), invoice_id INTEGER NOT NULL REFERENCES invoices(id), amount_cents INTEGER NOT NULL CHECK(amount_cents BETWEEN 1 AND 50000), paid_on TEXT NOT NULL) STRICT;
